Hard Java Tracing Questions

15 difficult exam-style Java tracing questions. For each question, write the exact output or write Compile-Time Error / Runtime Error.

How to Solve

Trace line by line. Pay attention to constructor chaining, static vs instance variables, method overriding, method overloading, exceptions, ArrayList removal, autoboxing, and String comparison.

class MyException extends Exception {
    MyException(String msg) {
        super(msg);
    }
}

class Check {
    static void validate(int x) throws MyException {
        if (x < 0)
            throw new MyException("Negative");
        else if (x == 0)
            throw new ArithmeticException("Zero");
        else
            System.out.println("Valid");
    }
}

    public static void main(String[] args) {
        try {
            Check.validate(-5);
            Check.validate(0);
            Check.validate(10);
        }
        catch (ArithmeticException e) {
            System.out.println("Arithmetic: " + e.getMessage());
        }
        catch (MyException e) {
            System.out.println("Custom: " + e.getMessage());
        }
        finally {
            System.out.println("Finally block");
        }

        System.out.println("After try");
    }
}
